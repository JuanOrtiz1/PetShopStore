CREATE TABLE `sac`.`sac_tb_product` (
  `PRODUCT` INT NOT NULL AUTO_INCREMENT,
  `DESCRIPTION` VARCHAR(100) NOT NULL,
  `SUPPLIER` VARCHAR(50) NOT NULL,
  `HIGH` INT NOT NULL,
  `TYPE` VARCHAR(100) NOT NULL,
  PRIMARY KEY (`PRODUCT`))
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8
COLLATE = utf8_spanish_ci
COMMENT = 'Tabla para listar los productos que existen en el punto de venta.';

CREATE TABLE `sac`.`sac_tb_supplier` (
  `SUPPLIER` INT NOT NULL AUTO_INCREMENT,
  `DESCRIPTION` VARCHAR(100) NOT NULL,
  `NUMBER_PHONE` VARCHAR(50) NOT NULL,
  `EMAIL` VARCHAR(100) NOT NULL,
  PRIMARY KEY (`SUPPLIER`))
)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8
COLLATE = utf8_spanish_ci
COMMENT = 'Tabla de proveedores.';

ALTER TABLE sac.sac_tb_supplier ADD COLUMN STATUS_SUPPLIER CHAR(1);



CREATE TABLE `sac`.`sac_tb_product` (
  `PRODUCT` INT AUTO_INCREMENT PRIMARY KEY,
  `NAME_PRODUCT` VARCHAR(255),
  `CATEGORY` VARCHAR(255),
  `PRESENTATION` VARCHAR(50),
  `PARAM_SUPPLIER` INT,
  FOREIGN KEY (`PARAM_SUPPLIER`) REFERENCES `sac_tb_supplier`(`SUPPLIER`),
  `PRODUCT_WEIGHT` INT(7),
  `PRODUCT_WEIGHT_TYPE` VARCHAR(20),
  `PURCHASE_PRICE` INT(7),
  `SALE_PRICE` INT(7),
  `REVENUE` INT(7),
  `STATUS` VARCHAR(2)
);
COMMENT = 'Tabla de productos';

CREATE TABLE sac_tb_stock (
    STOCK INT PRIMARY KEY AUTO_INCREMENT,
    PRODUCT INT,
    AMOUNT DECIMAL(10, 2) NOT NULL, -- Cantidad en unidades
    CONTAINER_TYPE VARCHAR(50), -- Tipo de envase
    UNIT_OF_MASS VARCHAR(20),
    STATUS_STOCK ENUM('Sellado', 'Destapado') NOT NULL, -- Estado del empaque
    CONTAINER_WEIGHT decimal(10,2), -- Tamaño del envase, si aplica
    DATE_ADMISSION DATE, -- Fecha en la que el stock llegó
    DUE_DATE DATE, -- Fecha de vencimiento, si aplica
    FOREIGN KEY (PRODUCT) REFERENCES sac_tb_product(PRODUCT)
);

CREATE TABLE `sac`.`sac_tb_sale` (
  `SALE` INT NOT NULL AUTO_INCREMENT,
  `STOCK` INT NOT NULL,
  `TYPE_SALE`  ENUM('Granel', 'Bulto', 'Arena', 'Accesorio', 'Paquete', 'Snack') NOT NULL,
  `SELLING_WEIGHT` INT,
  `SALE_PRICE` INT NOT NULL,
  `PURCHASE_PRICE` INT NOT NULL,
  `REVENUE` VARCHAR(100) NOT NULL,
  `PAYMENT_METHOD`ENUM('Efectivo', 'Nequi', 'Daviplata'),
  `MONEY_RECEIVED`  INT NOT NULL,
  `MONEY_RETURNED`  INT NOT NULL,
  `DATE_SALE` DATETIME NOT NULL,  
   `STATUS_SALE` VARCHAR(2) NOT NULL,
  FOREIGN KEY (STOCK) REFERENCES sac_tb_stock(STOCK),
  PRIMARY KEY (`SALE`))
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8
COLLATE = utf8_spanish_ci
COMMENT = 'Tabla para listar los productos que se venden en la tienda.';commit;
